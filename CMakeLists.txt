cmake_minimum_required(VERSION 3.20)
project(CreamyKeys VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
find_package(nlohmann_json 3.9 REQUIRED)
pkg_check_modules(ARCHIVE REQUIRED IMPORTED_TARGET libarchive)
pkg_check_modules(PW REQUIRED IMPORTED_TARGET libpipewire-0.3)
pkg_check_modules(EV REQUIRED IMPORTED_TARGET libevdev)
pkg_check_modules(SF REQUIRED IMPORTED_TARGET sndfile)
add_library(creamy-core src/core.cpp src/mechvibes.cpp)
target_link_libraries(creamy-core PUBLIC PkgConfig::SF PkgConfig::ARCHIVE nlohmann_json::nlohmann_json)
target_include_directories(creamy-core PUBLIC src)
target_compile_options(creamy-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(creamykeys src/main.cpp)
target_link_libraries(creamykeys PRIVATE creamy-core PkgConfig::PW PkgConfig::EV pthread)
target_compile_options(creamykeys PRIVATE -Wall -Wextra -Wpedantic)
add_executable(creamykeys-generate tools/generate.cpp)
target_link_libraries(creamykeys-generate PRIVATE PkgConfig::SF)
include(CTest)
if(BUILD_TESTING)
 add_executable(input-test tests/input-test.cpp)
 target_include_directories(input-test PRIVATE src)
 target_link_libraries(input-test PRIVATE PkgConfig::EV)
 add_test(NAME input-devices COMMAND input-test)
 add_executable(core-test tests/core-test.cpp)
 target_link_libraries(core-test PRIVATE creamy-core pthread)
 add_test(NAME core COMMAND core-test ${CMAKE_SOURCE_DIR}/soundpacks/creamy)
 add_executable(import-test tests/import-test.cpp)
 target_link_libraries(import-test PRIVATE creamy-core)
 add_test(NAME mechvibes-import COMMAND import-test ${CMAKE_SOURCE_DIR}/soundpacks/smooth)
endif()
install(TARGETS creamykeys RUNTIME DESTINATION bin)
install(DIRECTORY soundpacks/ DESTINATION share/creamykeys/packs)
